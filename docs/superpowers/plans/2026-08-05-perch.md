# Perch Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a native macOS floating mini-window ("Perch") that shows which Claude Code sessions are waiting on input across all terminal hosts, alerts on transition, and jumps to the terminal on click.

**Architecture:** Claude Code hooks write one atomic JSON status file per session to `~/.claude/perch/`. A SwiftUI/AppKit menu-bar app watches that directory, renders a non-activating floating `NSPanel` that hovers over everything (including fullscreen Spaces), fires a notification when a session transitions working→waiting, and dispatches a per-host "jump to terminal" command on click. cmux is the first-class host (jump via its socket CLI using surface/workspace IDs from cmux's own registry).

**Tech Stack:** Swift 6.3 (SwiftPM executable, no Xcode), AppKit + SwiftUI, UserNotifications, FSEvents; POSIX sh hooks; python3 for the settings installer.

## Global Constraints

- **Platform:** macOS only. Swift toolchain only — **no Xcode** (build via `swift build`, wrap in a `.app` with a script, ad-hoc `codesign`).
- **App type:** `LSUIElement` (menu-bar app, no Dock icon).
- **Status feed dir:** `~/.claude/perch/` — one file per session named `<session_id>.json`, schema version `1`.
- **State values:** exactly `working` | `waiting` | `ended`.
- **Atomic writes:** every hook writes a temp file then `mv` into place (never a partial JSON visible to the watcher).
- **Non-clobbering installer:** merge Perch hooks into existing `~/.claude/settings.json` `hooks` block; never drop the user's existing `gsd-*` hooks.
- **cmux jump commands (verbatim):** `cmux select-workspace --workspace <workspaceId>` then `cmux focus-panel --panel <surfaceId> --workspace <workspaceId>`.
- **cmux registry (read-only):** `~/.cmuxterm/claude-hook-sessions.json`, key `sessions[<session_id>]` → `{ surfaceId, workspaceId, cwd, pid }`.
- **Dismissal keying:** `(session_id, waiting_since)` in the app's `UserDefaults`; never written to session files.
- **Notifications:** edge-triggered on `working → waiting` only; never on reload of an already-waiting session.

```swift
protocol PidChecker { func isAlive(_ pid: Int32) -> Bool }
struct LivePidChecker: PidChecker {
    func isAlive(_ pid: Int32) -> Bool {
        if pid <= 0 { return false }
        return kill(pid, 0) == 0 || errno == EPERM
    }
}
protocol Clock { func now() -> Double }
struct LiveClock: Clock { func now() -> Double { Date().timeIntervalSince1970 } }
```

- [ ] **Step 4: Implement `SessionStore.swift`**

```swift
import Foundation
import Combine

final class SessionStore: ObservableObject {
    @Published private(set) var sessions: [SessionStatus] = []
    private let dir: URL
    private let pids: PidChecker
    private let clock: Clock
    private let defaultsKey = "PerchDismissedEpisodes"

    init(dir: URL, pids: PidChecker = LivePidChecker(), clock: Clock = LiveClock()) {
        self.dir = dir; self.pids = pids; self.clock = clock
    }

    private var dismissed: Set<String> {
        get { Set(UserDefaults.standard.stringArray(forKey: defaultsKey) ?? []) }
        set { UserDefaults.standard.set(Array(newValue), forKey: defaultsKey) }
    }

    func isDismissed(_ s: SessionStatus) -> Bool { dismissed.contains(s.episodeKey) }
    func dismiss(_ s: SessionStatus) { dismissed.insert(s.episodeKey); reload() }

    func reload() {
        let files = (try? FileManager.default.contentsOfDirectory(at: dir, includingPropertiesForKeys: nil)) ?? []
        var loaded: [SessionStatus] = []
        for f in files where f.pathExtension == "json" {
            guard let data = try? Data(contentsOf: f),
                  let s = try? SessionStatus.decode(from: data) else { continue }   // skip half-written/corrupt
            if s.state == .ended { continue }
            if !pids.isAlive(s.pid) { continue }
            if dismissed.contains(s.episodeKey) { continue }
            loaded.append(s)
        }
        DispatchQueue.main.async { self.sessions = self.sorted(loaded) }
    }

    private func sorted(_ xs: [SessionStatus]) -> [SessionStatus] {
        func rank(_ s: SessionStatus) -> Int { s.state == .waiting ? 0 : (s.state == .working ? 1 : 2) }
        return xs.sorted { a, b in
            if rank(a) != rank(b) { return rank(a) < rank(b) }
            if a.state == .waiting { return (a.waitingSince ?? 0) < (b.waitingSince ?? 0) }
            return a.lastActivity > b.lastActivity
        }
    }
}
```
Note: `@Published` assignment is dispatched to main; in tests call `reload()` synchronously — change the assignment to run synchronously when already on the main thread. Implement `reload()` to set `self.sessions = sorted(loaded)` directly (synchronous) and have the FSEvents caller hop to main before calling. Adjust the test to call on main (XCTest runs on main by default) — so keep `reload()` synchronous: replace the `DispatchQueue.main.async` line with `self.sessions = sorted(loaded)`.

```swift
struct TransitionDetector {
    private var seenWaitingEpisodes: Set<String> = []
    mutating func newlyWaiting(_ sessions: [SessionStatus]) -> [SessionStatus] {
```

```swift
import Foundation

enum JumpCommand: Equatable {
    case shell([String])
    case appleScript(String)
    case sequence([JumpCommand])
    case none(String)
}

struct JumpDispatcher {
    let cmux: CmuxLookup

    func command(for s: SessionStatus) -> JumpCommand {
        switch s.host {
        case "cmux":
            guard let surf = cmux.surface(for: s.sessionId) else { return .shell(["open", "-a", "cmux"]) }
            return .sequence([
                .shell(["cmux", "select-workspace", "--workspace", surf.workspaceId]),
                .shell(["cmux", "focus-panel", "--panel", surf.surfaceId, "--workspace", surf.workspaceId])
            ])
        case "iterm2":
            let sid = s.itermSessionId ?? ""
            let script = """
            tell application "iTerm2"
              activate
              repeat with w in windows
                repeat with t in tabs of w
                  repeat with ss in sessions of t
                    if id of ss is "\(sid)" then
                      select w
                      select t
                      select ss
                    end if
                  end repeat
                end repeat
              end repeat
            end tell
            """
            return .appleScript(script)
        case "tmux":
            guard let pane = s.tmuxPane else { return .none("no tmux pane recorded") }
            return .shell(["tmux", "select-pane", "-t", pane])
        case "vscode":
            return .shell(["code", s.cwd])
        default:
            return .none("unknown host \(s.host)")
        }
    }

    @discardableResult
    func run(_ c: JumpCommand, dryRun: Bool) -> [String] {
        switch c {
        case .shell(let args):
            let line = args.joined(separator: " ")
            if !dryRun { runProcess(args) }
            return [line]
        case .appleScript(let src):
            if !dryRun { runProcess(["osascript", "-e", src]) }
            return ["osascript"]
        case .sequence(let cmds):
            return cmds.flatMap { run($0, dryRun: dryRun) }
        case .none(let why):
            return ["# skipped: \(why)"]
        }
    }

    private func runProcess(_ args: [String]) {
        let p = Process()
        p.executableURL = URL(fileURLWithPath: "/usr/bin/env")
        p.arguments = args
        try? p.run()
    }
}
```

```swift
import Foundation

final class DirectoryWatcher {
    private var source: DispatchSourceFileSystemObject?
    private let fd: Int32
    private let onChange: () -> Void
    private var debounce: DispatchWorkItem?
    private let pollTimer: DispatchSourceTimer

    init(url: URL, onChange: @escaping () -> Void) {
        self.onChange = onChange
        fd = open(url.path, O_EVTONLY)
        if fd >= 0 {
            let s = DispatchSource.makeFileSystemObjectSource(fileDescriptor: fd, eventMask: [.write, .extend, .link], queue: .main)
            s.setEventHandler { [weak self] in self?.fire() }
            s.setCancelHandler { [weak self] in if let fd = self?.fd, fd >= 0 { close(fd) } }
            s.resume()
            source = s
        }
        // safety-net poll (covers atomic mv replacing the dir inode on some FSs)
        pollTimer = DispatchSource.makeTimerSource(queue: .main)
        pollTimer.schedule(deadline: .now() + 2, repeating: 2)
        pollTimer.setEventHandler { [weak self] in self?.onChange() }
        pollTimer.resume()
    }
    private func fire() {
        debounce?.cancel()
        let w = DispatchWorkItem { [weak self] in self?.onChange() }
        debounce = w
        DispatchQueue.main.asyncAfter(deadline: .now() + 0.15, execute: w)
    }
    deinit { source?.cancel(); pollTimer.cancel() }
}
```

```bash
#!/usr/bin/env bash
# Shared helpers for Perch hooks. Requires python3 (always present on macOS).
PERCH_DIR="${PERCH_DIR:-$HOME/.claude/perch}"

perch_detect_host() {
  if [ -n "${CMUX_SURFACE_ID:-}" ] || [ -n "${CMUX_WORKSPACE_ID:-}" ]; then echo cmux; return; fi
  case "${TERM_PROGRAM:-}" in
    iTerm.app) echo iterm2; return;;
    vscode)    echo vscode; return;;
  esac
  if [ -n "${TMUX:-}" ]; then echo tmux; return; fi
  echo unknown
}

# perch_write <session_id> <state> <cwd> <reason>
# Reads/merges nothing; recomputes the full file each event from env + args.
perch_write() {
  local sid="$1" state="$2" cwd="$3" reason="${4:-}"
  mkdir -p "$PERCH_DIR"
  local host; host="$(perch_detect_host)"
  local project; project="$(basename "$cwd")"
  local tmp; tmp="$(mktemp "$PERCH_DIR/.$sid.XXXXXX")"
  SID="$sid" STATE="$state" CWD="$cwd" PROJECT="$project" REASON="$reason" \
  HOST="$host" NOW="$(date +%s)" PID="${PERCH_PID:-$PPID}" \
  ITERM="${ITERM_SESSION_ID:-}" TMUXPANE="${TMUX_PANE:-}" TTY="$(tty 2>/dev/null || echo)" \
  python3 - "$tmp" <<'PY'
import json, os, sys
tmp = sys.argv[1]
prev = {}
final = os.path.join(os.environ["PERCH_DIR"] if False else os.path.dirname(tmp), os.environ["SID"] + ".json")
try:
    with open(final) as f: prev = json.load(f)
except Exception: prev = {}
now = int(os.environ["NOW"]); state = os.environ["STATE"]
d = {
  "schema": 1,
  "session_id": os.environ["SID"],
  "cwd": os.environ["CWD"],
  "project": os.environ["PROJECT"],
  "pid": int(os.environ["PID"] or 0),
  "state": state,
  "started_at": prev.get("started_at", now),
  "last_activity": now,
  "host": os.environ["HOST"],
  "iterm_session_id": os.environ.get("ITERM") or None,
  "tmux_pane": os.environ.get("TMUXPANE") or None,
  "tty": os.environ.get("TTY") or None,
}
if state == "waiting":
    d["waiting_since"] = prev.get("waiting_since", now) if prev.get("state") == "waiting" else now
    if os.environ.get("REASON"): d["reason"] = os.environ["REASON"]
with open(tmp, "w") as f: json.dump(d, f)
PY
  mv -f "$tmp" "$PERCH_DIR/$sid.json"
}

# Extract a field from the hook's stdin JSON payload.
perch_stdin_field() {
  python3 -c 'import json,sys; d=json.load(sys.stdin); print(d.get(sys.argv[1],""))' "$1"
}
```

## Task 8: Hook installer (non-clobbering)

Merge Perch's hooks into `~/.claude/settings.json` without dropping existing hooks.

**Files:**
- Create: `scripts/install-hooks.sh`

**Interfaces:**
- Consumes: `hooks/*.sh` (installed to `~/.claude/hooks/perch/`).
- Produces: updated `~/.claude/settings.json` with Perch commands appended to the `SessionStart`, `UserPromptSubmit`, `Stop`, `Notification`, `SessionEnd` hook arrays.

- [ ] **Step 1: Write `scripts/install-hooks.sh`**

```bash
#!/usr/bin/env bash
set -euo pipefail
cd "$(dirname "$0")/.."
DEST="$HOME/.claude/hooks/perch"
mkdir -p "$DEST"
cp hooks/*.sh "$DEST/"; chmod +x "$DEST"/*.sh
SETTINGS="$HOME/.claude/settings.json"
[ -f "$SETTINGS" ] && cp "$SETTINGS" "$SETTINGS.perch.bak.$(date +%s)"

DEST="$DEST" python3 - "$SETTINGS" <<'PY'
import json, os, sys
path = sys.argv[1]; dest = os.environ["DEST"]
try:
    with open(path) as f: cfg = json.load(f)
except Exception: cfg = {}
hooks = cfg.setdefault("hooks", {})
mapping = {
  "SessionStart": "perch-session-start.sh",
  "UserPromptSubmit": "perch-user-prompt.sh",
  "Stop": "perch-stop.sh",
  "Notification": "perch-notification.sh",
  "SessionEnd": "perch-session-end.sh",
}
for event, script in mapping.items():
    cmd = f'bash "{dest}/{script}"'
    arr = hooks.setdefault(event, [])
    # Claude Code shape: [ { "hooks": [ { "type":"command", "command": "..." } ] } ]
    if any(cmd in json.dumps(entry) for entry in arr):
        continue
    arr.append({"hooks": [{"type": "command", "command": cmd}]})
with open(path, "w") as f: json.dump(cfg, f, indent=2)
print("Installed Perch hooks into", path)
PY
```
